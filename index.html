STEP 01

여러 CSV 파일 자동 통합

pathlib로 입력 폴더의 모든 CSV를 검색하고 csv.DictReader로 읽어 하나의 파일로 합칩니다.

Path.glob()

조건에 맞는 파일을 한꺼번에 찾습니다.

DictReader

열 이름을 키로 사용해 한 행씩 읽습니다.

DictWriter

통합 결과를 열 이름과 함께 저장합니다.

step1_merge.pyCSV 검색·통합
# 파일과 폴더 경로를 다루기 위한 Path를 가져옵니다.
from pathlib import Path

# CSV 파일을 읽고 쓰기 위한 csv 모듈을 가져옵니다.
import csv


def merge_csv_files():
    # 프로그램을 실행한 위치를 기준으로 input과 output 경로를 만듭니다.
    input_folder = Path("input")
    output_folder = Path("output")

    # output 폴더가 없으면 새로 만듭니다.
    # 이미 존재해도 exist_ok=True이므로 오류가 나지 않습니다.
    output_folder.mkdir(exist_ok=True)

    # input 폴더에서 확장자가 .csv인 파일을 모두 찾아 리스트로 만듭니다.
    csv_files = list(input_folder.glob("*.csv"))

    # CSV 파일이 한 개도 없으면 안내문을 출력하고 함수를 종료합니다.
    if not csv_files:
        print("input 폴더에 CSV 파일이 없습니다.")
        return

    # 모든 파일에서 읽은 거래내역을 모아 둘 빈 리스트입니다.
    merged_rows = []

    # 찾은 CSV 파일을 하나씩 반복해서 처리합니다.
    for csv_file in csv_files:
        # utf-8-sig는 엑셀에서 만든 한글 CSV의 첫 열이 깨지는 것을 줄여 줍니다.
        with open(csv_file, "r", encoding="utf-8-sig") as file:
            # 첫 행의 열 이름을 키로 사용하여 각 행을 딕셔너리로 읽습니다.
            reader = csv.DictReader(file)

            for row in reader:
                # 거래가 어느 파일에서 왔는지 알 수 있도록 원본 파일명을 추가합니다.
                row["원본파일"] = csv_file.name

                # 원본 파일명이 추가된 한 건의 거래를 통합 리스트에 넣습니다.
                merged_rows.append(row)

    # 결과 파일에 기록할 열의 이름과 순서를 지정합니다.
    fieldnames = ["원본파일", "거래일자", "거래처", "금액", "결제수단", "메모"]

    # / 연산자로 폴더 경로와 파일명을 연결합니다.
    output_path = output_folder / "merged_transactions.csv"

    # 통합 결과를 쓰기 모드로 저장합니다.
    # newline=""은 Windows에서 행 사이에 빈 줄이 생기는 것을 방지합니다.
    with open(output_path, "w", newline="", encoding="utf-8-sig") as file:
        writer = csv.DictWriter(file, fieldnames=fieldnames)

        # 첫 번째 행에 열 이름을 기록합니다.
        writer.writeheader()

        # merged_rows의 모든 거래내역을 한꺼번에 기록합니다.
        writer.writerows(merged_rows)

    # 처리한 파일 수, 거래 건수, 저장된 절대경로를 출력합니다.
    print(f"{len(csv_files)}개 파일, {len(merged_rows)}건을 통합했습니다.")
    print(f"저장 위치: {output_path.resolve()}")


# 위에서 정의한 함수를 호출하여 실제 통합 작업을 시작합니다.
merge_csv_files()

# 파일 경로 처리와 CSV 파일 입출력에 필요한 도구를 가져옵니다.
from pathlib import Path
import csv


# 거래처에 포함된 키워드와 계정과목의 대응 규칙입니다.
# 실제 업무에서는 회사의 회계처리 기준에 맞게 수정해야 합니다.
ACCOUNT_RULES = {
    "스타벅스": "복리후생비", "배달의민족": "복리후생비",
    "쿠팡": "소모품비", "알파문구": "소모품비",
    "코레일": "여비교통비", "한국전력": "수도광열비",
    "사무실임대": "임차료", "KT": "통신비",
    "네이버클라우드": "통신비", "국세청": "세금과공과",
    "세무법인": "지급수수료"
}


def classify_account(merchant):
    # 거래처명에 등록된 키워드가 들어 있는지 차례대로 검사합니다.
    for keyword, account in ACCOUNT_RULES.items():
        if keyword in merchant:
            # 일치하는 키워드를 찾으면 대응하는 계정과목을 반환합니다.
            return account
    # 일치하는 규칙이 없으면 임의로 정하지 않고 확인 대상으로 남깁니다.
    return "확인 필요"


def validate_row(row):
    # 거래일자와 거래처가 비어 있는지 검사합니다.
    if not row["거래일자"].strip():
        return "거래일자 누락"
    if not row["거래처"].strip():
        return "거래처 누락"

    # 금액을 정수로 바꿀 수 있는지, 0보다 큰지 검사합니다.
    try:
        amount = int(row["금액"])
        if amount <= 0:
            return "금액이 0 이하"
    except ValueError:
        # '삼만원'처럼 숫자로 바꿀 수 없는 값이 들어오면 실행됩니다.
        return "금액이 숫자가 아님"

    # 빈 문자열은 발견된 오류가 없다는 뜻입니다.
    return ""


def validate_and_classify():
    # 1단계 결과 파일과 2단계에서 만들 결과 파일의 경로입니다.
    input_path = Path("output/merged_transactions.csv")
    clean_path = Path("output/clean_transactions.csv")
    review_path = Path("output/review_transactions.csv")

    # 검증 통과 거래와 확인 필요 거래를 나누어 담습니다.
    clean_rows, review_rows = [], []

    # 1단계에서 통합한 CSV 파일을 한 행씩 읽습니다.
    with open(input_path, "r", encoding="utf-8-sig") as file:
        for row in csv.DictReader(file):
            # 한 건의 거래를 검사하고 오류 사유를 받습니다.
            error = validate_row(row)

            # 오류가 있으면 상태를 기록하고 확인 목록으로 보냅니다.
            if error:
                row["계정과목"] = ""
                row["처리상태"] = error
                review_rows.append(row)
                # 현재 행의 나머지 분류 작업을 건너뛰고 다음 거래로 이동합니다.
                continue

            # 검증을 통과한 금액을 계산 가능한 정수로 변환합니다.
            row["금액"] = int(row["금액"])

            # 거래처명으로 계정과목을 자동 분류합니다.
            row["계정과목"] = classify_account(row["거래처"])
            row["처리상태"] = "정상" if row["계정과목"] != "확인 필요" else "계정과목 확인"

            # 입력값 검증을 통과한 거래는 clean_rows에 저장합니다.
            clean_rows.append(row)

            # 계정과목을 정하지 못한 거래는 review_rows에도 복사합니다.
            if row["처리상태"] != "정상":
                review_rows.append(row.copy())

    # 두 결과 파일에 공통으로 사용할 열 이름입니다.
    fieldnames = ["원본파일", "거래일자", "거래처", "금액", "결제수단", "메모", "계정과목", "처리상태"]

    # 정상 파일과 확인 필요 파일을 같은 방식으로 각각 저장합니다.
    for path, rows in [(clean_path, clean_rows), (review_path, review_rows)]:
        with open(path, "w", newline="", encoding="utf-8-sig") as file:
            writer = csv.DictWriter(file, fieldnames=fieldnames)
            writer.writeheader()
            writer.writerows(rows)

    # 저장된 거래 건수를 화면에 출력합니다.
    print(f"검증 통과: {len(clean_rows)}건")
    print(f"확인 필요: {len(review_rows)}건")


# 검증과 자동 분류 작업을 시작합니다.
validate_and_classify()

# 파일 경로, 항목별 합계, CSV 읽기에 필요한 도구를 가져옵니다.
from pathlib import Path
from collections import defaultdict
import csv

# Excel 파일 생성과 서식 지정에 필요한 openpyxl 도구를 가져옵니다.
from openpyxl import Workbook
from openpyxl.styles import Font, PatternFill, Alignment
from openpyxl.utils import get_column_letter


def read_csv(file_path):
    # 전달받은 CSV 파일을 열어 모든 행을 리스트로 반환합니다.
    with open(file_path, "r", encoding="utf-8-sig") as file:
        return list(csv.DictReader(file))


def style_sheet(sheet, header_color):
    # 첫 번째 행의 셀에 배경색, 흰색 글자, 굵은 글자와 가운데 정렬을 적용합니다.
    for cell in sheet[1]:
        cell.fill = PatternFill("solid", fgColor=header_color)
        cell.font = Font(color="FFFFFF", bold=True)
        cell.alignment = Alignment(horizontal="center")

    # 각 열에서 가장 긴 내용에 맞춰 열 너비를 조절합니다.
    for column in sheet.columns:
        width = max(len(str(cell.value or "")) for cell in column) + 2
        # 지나치게 넓어지지 않도록 최대 너비는 30으로 제한합니다.
        sheet.column_dimensions[get_column_letter(column[0].column)].width = min(width, 30)

    # 스크롤해도 제목 행이 보이게 하고 전체 표에 자동 필터를 설정합니다.
    sheet.freeze_panes = "A2"
    sheet.auto_filter.ref = sheet.dimensions


def create_report():
    # 2단계에서 만든 정상 파일과 확인 필요 파일을 읽습니다.
    output_folder = Path("output")
    clean_rows = read_csv(output_folder / "clean_transactions.csv")
    review_rows = read_csv(output_folder / "review_transactions.csv")

    # Excel 시트에서 사용할 열 이름과 순서를 정합니다.
    headers = ["원본파일", "거래일자", "거래처", "금액", "결제수단", "메모", "계정과목", "처리상태"]

    # 새 Excel 통합문서를 만들고 기본 시트의 이름을 변경합니다.
    workbook = Workbook()
    all_sheet = workbook.active
    all_sheet.title = "전체 거래내역"
    all_sheet.append(headers)

    # 검증을 통과한 거래를 전체 거래내역 시트에 기록합니다.
    for row in clean_rows:
        # Excel에서 합계를 계산할 수 있도록 금액을 정수로 변환합니다.
        row["금액"] = int(row["금액"])
        all_sheet.append([row[header] for header in headers])

    # 존재하지 않는 키에 처음 접근하면 0으로 시작하는 딕셔너리입니다.
    totals = defaultdict(int)
    counts = defaultdict(int)

    # 계정과목별 금액 합계와 거래 건수를 계산합니다.
    for row in clean_rows:
        totals[row["계정과목"]] += row["금액"]
        counts[row["계정과목"]] += 1

    # 계정과목별 합계 시트를 만들고 계산 결과를 기록합니다.
    summary_sheet = workbook.create_sheet("계정과목별 합계")
    summary_sheet.append(["계정과목", "거래 건수", "합계 금액"])
    for account in sorted(totals):
        summary_sheet.append([account, counts[account], totals[account]])
    summary_sheet.append(["전체 합계", sum(counts.values()), sum(totals.values())])

    # 입력 오류 또는 분류 확인이 필요한 거래만 모은 시트를 만듭니다.
    review_sheet = workbook.create_sheet("확인 필요 거래")
    review_sheet.append(headers)
    for row in review_rows:
        review_sheet.append([row.get(header, "") for header in headers])

    # 각 시트에 서로 다른 제목 색상과 공통 표 서식을 적용합니다.
    style_sheet(all_sheet, "1D4ED8")
    style_sheet(summary_sheet, "6D28D9")
    style_sheet(review_sheet, "EA580C")

    # 금액 셀을 1,000처럼 천 단위 쉼표가 표시되도록 설정합니다.
    for cell in all_sheet["D"][1:]:
        cell.number_format = "#,##0"
    for cell in summary_sheet["C"][1:]:
        cell.number_format = "#,##0"

    # 완성된 통합문서를 output 폴더에 Excel 파일로 저장합니다.
    report_path = output_folder / "거래내역_자동정리_보고서.xlsx"
    workbook.save(report_path)
    print(f"보고서가 생성되었습니다: {report_path.resolve()}")


# Excel 보고서 생성 작업을 시작합니다.
create_report()

def detect_file_format(fieldnames):
    """CSV의 컬럼 조합을 검사하여 어느 기관 형식인지 판별합니다."""

    # DictReader가 읽은 컬럼명을 집합으로 바꾸면 포함 여부를 쉽게 검사할 수 있습니다.
    file_columns = set(fieldnames or [])

    # 등록된 기관별 규칙을 하나씩 확인합니다.
    for format_name, column_map in COLUMN_RULES.items():
        required_columns = set(column_map.keys())

        # 현재 파일이 해당 기관의 필수 컬럼을 모두 가지고 있으면 그 규칙을 반환합니다.
        # 원본 파일에 추가 컬럼이 더 있어도 필수 컬럼만 있으면 처리할 수 있습니다.
        if required_columns.issubset(file_columns):
            return format_name, column_map

    # 어떤 규칙과도 맞지 않으면 억지로 합치지 않고 오류를 발생시킵니다.
    raise ValueError(
        "등록되지 않은 컬럼 구조입니다: "
        + ", ".join(fieldnames or [])
    )


def normalize_date(value):
    """여러 형태의 날짜를 YYYY-MM-DD 형식으로 통일합니다."""

    value = value.strip()

    # 실습파일에서 사용한 날짜 형식을 차례대로 대입해 봅니다.
    date_formats = [
        "%Y-%m-%d",
        "%Y.%m.%d",
        "%Y/%m/%d",
        "%Y%m%d",
        "%Y%m%d %H:%M",
    ]

    for date_format in date_formats:
        try:
            parsed_date = datetime.strptime(value, date_format)
            return parsed_date.strftime("%Y-%m-%d")
        except ValueError:
            # 현재 형식과 맞지 않으면 다음 날짜 형식으로 다시 시도합니다.
            continue

    raise ValueError(f"지원하지 않는 날짜 형식: {value}")


def normalize_amount(value):
    """금액의 쉼표·원·공백을 제거하고 계산 가능한 정수로 바꿉니다."""

    cleaned_value = (
        value.strip()
        .replace(",", "")
        .replace("원", "")
        .replace(" ", "")
    )

    try:
        return int(cleaned_value)
    except ValueError as error:
        raise ValueError(f"금액을 숫자로 바꿀 수 없음: {value}") from error


def classify_account(merchant):
    """거래처 키워드를 이용하여 계정과목을 반환합니다."""

    for keyword, account in ACCOUNT_RULES.items():
        if keyword in merchant:
            return account

    return "확인 필요"


def standardize_row(row, file_name, format_name, column_map):
    """서로 다른 원본 행을 하나의 표준 거래내역으로 변환합니다."""

    standardized = {}

    # 원본 컬럼의 값을 가져와 대응하는 표준 컬럼명으로 저장합니다.
    for original_column, standard_column in column_map.items():
        standardized[standard_column] = row.get(original_column, "").strip()

    # 필수값이 비어 있으면 잘못된 자료가 섞이지 않도록 중단합니다.
    if not standardized["거래일자"]:
        raise ValueError("거래일자 누락")
    if not standardized["거래처"]:
        raise ValueError("거래처 누락")
    if not standardized["금액"]:
        raise ValueError("금액 누락")

    # 컬럼 이름뿐 아니라 날짜와 금액의 값 형식도 통일합니다.
    standardized["거래일자"] = normalize_date(standardized["거래일자"])
    standardized["금액"] = normalize_amount(standardized["금액"])

    # 표준화된 거래처명을 이용해 기존 분류 규칙을 적용합니다.
    account = classify_account(standardized["거래처"])
    status = "정상" if account != "확인 필요" else "계정과목 확인"

    # 최종 결과는 모든 원본 파일에서 동일한 컬럼과 순서를 갖습니다.
    return {
        "원본파일": file_name,
        "원본형식": format_name,
        "거래일자": standardized["거래일자"],
        "거래처": standardized["거래처"],
        "금액": standardized["금액"],
        "결제수단": standardized["결제수단"],
        "메모": standardized["메모"],
        "계정과목": account,
        "처리상태": status,
    }


def load_and_standardize_files():
    """advanced_input의 모든 CSV를 읽어 표준 거래내역으로 통합합니다."""

    csv_files = sorted(INPUT_FOLDER.glob("*.csv"))

    if not csv_files:
        print(f"CSV 파일이 없습니다: {INPUT_FOLDER}")
        return []

    all_rows = []

    for csv_file in csv_files:
        with open(csv_file, "r", encoding="utf-8-sig") as file:
            reader = csv.DictReader(file)

            # 파일명이 아니라 실제 컬럼 조합으로 기관 형식을 판별합니다.
            format_name, column_map = detect_file_format(reader.fieldnames)
            print(f"{csv_file.name} → {format_name} 형식으로 판별")

            for line_number, row in enumerate(reader, start=2):
                try:
                    standardized_row = standardize_row(
                        row,
                        csv_file.name,
                        format_name,
                        column_map,
                    )
                    all_rows.append(standardized_row)
                except ValueError as error:
                    # 오류가 난 행과 이유를 화면에 알려 주고 다음 행을 계속 처리합니다.
                    print(f"  {line_number}행 제외: {error}")

    # 파일 종류와 관계없이 최종 결과를 거래일자순으로 정렬합니다.
    all_rows.sort(key=lambda row: (row["거래일자"], row["원본파일"]))
    return all_rows


def save_standard_csv(rows):
    """표준화한 전체 거래를 하나의 CSV 파일로 저장합니다."""

    csv_path = OUTPUT_FOLDER / "standardized_transactions.csv"

    with open(
        csv_path,
        "w",
        newline="",
        encoding="utf-8-sig",
    ) as file:
        writer = csv.DictWriter(file, fieldnames=STANDARD_COLUMNS)
        writer.writeheader()
        writer.writerows(rows)

    return csv_path


def style_sheet(sheet, header_color):
    """Excel 시트의 제목과 열 너비에 공통 서식을 적용합니다."""

    for cell in sheet[1]:
        cell.fill = PatternFill("solid", fgColor=header_color)
        cell.font = Font(color="FFFFFF", bold=True)
        cell.alignment = Alignment(horizontal="center")

    for column in sheet.columns:
        width = max(len(str(cell.value or "")) for cell in column) + 2
        sheet.column_dimensions[get_column_letter(column[0].column)].width = min(
            width,
            30,
        )

    sheet.freeze_panes = "A2"
    sheet.auto_filter.ref = sheet.dimensions


def create_excel_report(rows):
    """표준 거래·계정과목별 합계·확인 필요 시트가 있는 보고서를 만듭니다."""

    workbook = Workbook()

    # 첫 번째 시트에는 표준화된 전체 거래를 기록합니다.
    transaction_sheet = workbook.active
    transaction_sheet.title = "표준 거래내역"
    transaction_sheet.append(STANDARD_COLUMNS)

    for row in rows:
        transaction_sheet.append([row[column] for column in STANDARD_COLUMNS])

    # 두 번째 시트에는 계정과목별 거래 건수와 합계를 기록합니다.
    totals = defaultdict(int)
    counts = defaultdict(int)

    for row in rows:
        totals[row["계정과목"]] += row["금액"]
        counts[row["계정과목"]] += 1

    summary_sheet = workbook.create_sheet("계정과목별 합계")
    summary_sheet.append(["계정과목", "거래 건수", "합계 금액"])

    for account in sorted(totals):
        summary_sheet.append([account, counts[account], totals[account]])

    summary_sheet.append(
        ["전체 합계", sum(counts.values()), sum(totals.values())]
    )

    # 세 번째 시트에는 사람이 계정과목을 확인해야 할 거래만 기록합니다.
    review_sheet = workbook.create_sheet("확인 필요 거래")
    review_sheet.append(STANDARD_COLUMNS)

    for row in rows:
        if row["처리상태"] != "정상":
            review_sheet.append([row[column] for column in STANDARD_COLUMNS])

    style_sheet(transaction_sheet, "1D4ED8")
    style_sheet(summary_sheet, "6D28D9")
    style_sheet(review_sheet, "EA580C")

    # 금액 열에는 천 단위 쉼표 서식을 적용합니다.
    for cell in transaction_sheet["E"][1:]:
        cell.number_format = "#,##0"
    for cell in summary_sheet["C"][1:]:
        cell.number_format = "#,##0"

    report_path = OUTPUT_FOLDER / "different_columns_report.xlsx"
    workbook.save(report_path)
    return report_path


def main():
    """컬럼 판별부터 표준화·통합·보고서 생성까지 한 번에 실행합니다."""

    OUTPUT_FOLDER.mkdir(exist_ok=True)
    rows = load_and_standardize_files()

    if not rows:
        return

    csv_path = save_standard_csv(rows)
    report_path = create_excel_report(rows)
    total_amount = sum(row["금액"] for row in rows)

    print()
    print(f"표준화 완료: {len(rows)}건")
    print(f"전체 합계: {total_amount:,}원")
    print(f"CSV 저장: {csv_path}")
    print(f"Excel 저장: {report_path}")


if __name__ == "__main__":
    main()

새롭게 추가된 부분 이해하기

1. 기관별 컬럼 변환 규칙 등록

COLUMN_RULES에는 각 기관의 원본 컬럼과 프로그램에서 사용할 표준 컬럼의 대응 관계가 들어 있습니다. 예를 들어 카드사 A의 이용일, 은행의 거래일시, 카드사 B의 승인일자는 모두 거래일자로 바뀝니다.

2. 파일명이 아니라 실제 컬럼 조합으로 형식 판별

detect_file_format()은 파일의 첫 행에 있는 컬럼명을 읽고 등록된 규칙과 비교합니다. 필요한 컬럼을 모두 가지고 있으면 해당 기관의 형식으로 판별합니다. 파일명이 바뀌어도 컬럼 구조가 같으면 처리할 수 있습니다.

3. 컬럼명과 값 형식을 함께 표준화

standardize_row()은 원본 컬럼의 값을 표준 컬럼으로 옮깁니다. 컬럼 이름만 바꾸는 것으로 끝내지 않고 normalize_date()와 normalize_amount()를 호출하여 날짜와 금액도 계산 가능한 형태로 바꿉니다.

4. 서로 다른 날짜 형식 통일

2026.10.02, 2026/10/06, 20261004, 20261005 09:30처럼 표현이 다른 날짜를 모두 2026-10-02와 같은 YYYY-MM-DD 형식으로 저장합니다.

5. 문자처럼 보이는 금액을 정수로 변환

12,500, 35,000원, 88000에서 쉼표·원·공백을 제거한 뒤 정수로 변환합니다. 이 과정을 거쳐야 계정과목별 합계와 전체 합계를 정확하게 계산할 수 있습니다.

6. 원본 추적 정보 유지

표준화가 끝나도 원본파일과 원본형식을 결과에 남깁니다. 결과값이 이상할 때 어떤 기관의 어느 파일에서 들어온 거래인지 다시 확인할 수 있습니다.

7. 표준화 이후에는 기존 처리 방식 재사용

모든 거래가 거래일자·거래처·금액·결제수단·메모 구조로 통일된 다음에는 앞에서 학습한 계정과목 분류와 Excel 보고서 생성 방식을 그대로 적용할 수 있습니다. 새로 추가된 핵심은 기존 처리 앞에 놓인 형식 판별과 표준화 과정입니다.

반드시 확인할 한계: 이 코드는 등록된 세 가지 컬럼 구조만 처리합니다. 새로운 카드사나 은행 파일이 들어오면 COLUMN_RULES에 그 기관의 규칙을 추가해야 합니다. 특히 출금액, 입금액, 취소금액은 이름이 비슷해도 회계적 의미가 다르므로 프로그램이 임의로 추측하게 만들면 안 됩니다.
